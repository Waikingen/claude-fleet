```python
#!/usr/bin/env python3
"""
Claude Fleet Daemon v2
Persistent async supervisor for Claude Code sessions.
Polls iTerm2 every 3s, analyzes paused sessions, writes state.json.
"""
from __future__ import annotations

import asyncio
import hashlib
import json
import os
import re
import signal
import subprocess
import sys
import tempfile
from dataclasses import dataclass, field, asdict
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional

try:
    import iterm2
except ImportError:
    print("ERROR: iterm2 not installed. Run: pip3 install iterm2", file=sys.stderr)
    sys.exit(1)

# ── Paths ──────────────────────────────────────────────────────────────────────
HOME       = Path.home()
FLEET_DIR  = HOME / ".claude-fleet"
STATE_PATH = FLEET_DIR / "state.json"
ROLES_PATH = FLEET_DIR / "roles.json"
LOG_PATH   = FLEET_DIR / "history.jsonl"
SOCK_PATH  = FLEET_DIR / "fleet.sock"
PROMPT_PATH = FLEET_DIR / "analyze_prompt.txt"
FLEET_DIR.mkdir(parents=True, exist_ok=True)

ANALYSIS_TIMEOUT = 15 # seconds per claude call

# ── Pause detection patterns ───────────────────────────────────────────────────
PAUSE_PATTERNS = [
    re.compile(r"\?\s.+\(y/n\)", re.I),
    re.compile(r"\?\s.+\[y/n\]", re.I),
    re.compile(r"\?\s.+\[Y/n\]"),
    re.compile(r"Do you want to", re.I),
    re.compile(r"Would you like to", re.I),
    re.compile(r"Allow\b.{0,80}\?", re.I),
    re.compile(r"Approve\b.{0,80}\?", re.I),
    re.compile(r"\[1\].{0,150}\[2\]", re.DOTALL),
    re.compile(r"❯\s+[Yy]es"),
    re.compile(r"❯\s+[Nn]o"),
    re.compile(r"Allow .{0,60}\("),
    re.compile(r"Run command\?", re.I),
    re.compile(r"Overwrite\b.{0,40}\?", re.I),
]

RISKY_PATTERNS = [
    (re.compile(r"drop\s+table\b", re.I),     "DROP TABLE"),
    (re.compile(r"delete\s+from\b", re.I),    "DELETE FROM (all rows)"),
    (re.compile(r"truncate\s+table\b", re.I), "TRUNCATE TABLE"),
    (re.compile(r"drop\s+database\b", re.I),  "DROP DATABASE"),
    (re.compile(r"\brm\s+-rf\b"),             "rm -rf"),
    (re.compile(r"force.?push", re.I),        "force push"),
    (re.compile(r"reset\s+--hard"),           "git reset --hard"),
    (re.compile(r"git\s+clean\s+-f"),         "git clean -f"),
    (re.compile(r"irreversible", re.I),       "irreversible operation"),
    (re.compile(r"permanently\s+delet", re.I),"permanent deletion"),
]

STRIP_ANSI = re.compile(r"\x1b\[[0-9;]*[mKJHABCDEFGsuhl]|\x1b\[?\d*[A-Za-z]|\r")

# ── Built-in roles ─────────────────────────────────────────────────────────────

# ── Data classes ───────────────────────────────────────────────────────────────
@dataclass
class SessionAnalysis:
    action: str = "continue"
    waiting_for: str = ""
    options: list = field(default_factory=lambda: ["y", "n"])
    recommended: Optional[str] = None
    confidence: float = 0.0
    risk: bool = False
    risk_reason: Optional[str] = None
    decision: Optional[str] = None
    auto_sent: Optional[str] = None
    ts_paused: Optional[str] = None

@dataclass
class SessionState:
    session_id: str
    title: str
    project: str
    branch: str
    path: str
    last_lines: str
    content_hash: str
    blocked: bool = False
    role: str = "builder"
    analysis: SessionAnalysis = field(default_factory=SessionAnalysis)

# ── Globals ────────────────────────────────────────────────────────────────────
session_cache: dict[str, SessionState] = {}  # session_id -> SessionState
roles_config: dict = {}
mode: str = "scout"  # scout | semi | autopilot
running: bool = True


# ── Roles ──────────────────────────────────────────────────────────────────────
def load_roles() -> dict:
    try:
        return json.loads(ROLES_PATH.read_text()) if ROLES_PATH.exists() else {}
    except Exception:
        return {}

def get_role_for_session(session_id: str, project: str) -> dict:
    """Priority: session > project > global > builder"""
    cfg = roles_config
    role_id = (
        cfg.get("sessions", {}).get(session_id)
        or cfg.get("projects", {}).get(project)
        or cfg.get("global", "builder")
    )
    custom = cfg.get("custom", {}).get(role_id)
    if custom:
        return custom
    return BUILTIN_ROLES.get(role_id, BUILTIN_ROLES["builder"])

def save_roles():
    tmp = ROLES_PATH.with_suffix(".tmp")
    tmp.write_text(json.dumps(roles_config, indent=2))
    tmp.rename(ROLES_PATH)


# ── Utilities ──────────────────────────────────────────────────────────────────
def strip_ansi(text: str) -> str:
    return STRIP_ANSI.sub("", text)

def content_hash(text: str) -> str:
    return hashlib.md5(text.encode()).hexdigest()

def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()

def is_paused(content: str) -> bool:
    return any(p.search(content) for p in PAUSE_PATTERNS)

def check_risk(content: str) -> tuple[bool, Optional[str]]:
    for pattern, label in RISKY_PATTERNS:
        if pattern.search(content):
            return True, label
    return False, None

def get_project_info(path: str) -> dict:
    p = Path(path)
    name = p.name
    branch = "main"
    try:
        branch = subprocess.check_output(
            ["git", "rev-parse", "--abbrev-ref", "HEAD"],
            cwd=path, capture_output=True if hasattr(subprocess, 'DEVNULL') else False,
            stderr=subprocess.DEVNULL, text=True, timeout=2
        ).strip() or "main"
    except Exception:
        pass
    return {"name": name, "branch": branch}

def log_action(session_id: str, project: str, action: str, response: str, auto: bool):
    entry = {
        "ts": now_iso(),
        "session_id": session_id,
        "project": project,
        "action": action,
        "response": response,
        "auto": auto,
    }
    with LOG_PATH.open("a") as f:
        f.write(json.dumps(entry) + "\n")


# ── State writer ───────────────────────────────────────────────────────────────
def write_state(phase: str = "done"):
    sessions_out = []
    paused = 0
    auto_resolved = 0
    needs_review = 0

    for s in session_cache.values():
        a = s.analysis
        out = {
            "session_id":  s.session_id,
            "title":       s.title,
            "project":     s.project,
            "branch":      s.branch,
            "path":        s.path,
            "last_lines":  s.last_lines,
            "role":        s.role,
            "action":      a.action,
            "waiting_for": a.waiting_for,
            "options":     a.options,
            "recommended": a.recommended,
            "confidence":  a.confidence,
            "risk":        a.risk,
            "risk_reason": a.risk_reason,
            "decision":    a.decision,
            "blocked":     s.blocked,
            "auto_sent":   a.auto_sent,
            "ts_paused":   a.ts_paused,
        }
        sessions_out.append(out)
        if a.action == "respond":
            paused += 1
            if a.auto_sent:
                auto_resolved += 1
            else:
                needs_review += 1

    # Sort: needs-review first, then pending, qa, continue
    order = {"respond": 0, "pending": 1, "qa": 2, "continue": 3}
    sessions_out.sort(key=lambda s: (order.get(s["action"], 9), s["project"]))

    state = {
        "phase":          phase,
        "mode":           mode,
        "sessions":       sessions_out,
        "total_sessions": len(sessions_out),
        "paused_count":   paused,
        "auto_resolved":  auto_resolved,
        "needs_review":   needs_review,
        "updated":        now_iso(),
    }

    tmp = STATE_PATH.with_suffix(".tmp")
    tmp.write_text(json.dumps(state, indent=2))
    tmp.rename(STATE_PATH)


async def main():
    print(f"[fleet] daemon starting, socket={SOCK_PATH}", flush=True)
    global roles_config
    roles_config = load_roles()
    write_state("scanning")

    async with iterm2.connect(app=None) as connection:
        print("[fleet] iTerm2 connected", flush=True)

        # Start socket server and poll loop concurrently
        server_task = asyncio.create_task(run_socket_server(connection))
        poll_task = asyncio.create_task(poll_loop(connection))

        await asyncio.gather(server_task, poll_task)


async def poll_loop(connection):
    """Main 3s poll loop — sweep iTerm2 sessions, analyze changed ones."""
    global running
    while running:
        try:
            await sweep(connection)
        except Exception as e:
            print(f"[fleet] poll error: {e}", flush=True)
        await asyncio.sleep(POLL_INTERVAL)


async def sweep(connection):
    """One full sweep of all iTerm2 sessions."""
    write_state("scanning")
    app = await iterm2.async_get_app(connection)
    all_session_ids = set()

    tasks = []
    for window in app.windows:
        for tab in window.tabs:
            for session in tab.sessions:
                all_session_ids.add(session.session_id)
                tasks.append(process_session(session))

    if tasks:
        await asyncio.gather(*tasks, return_exceptions=True)

    # Remove sessions that no longer exist in iTerm2
    for gone in set(session_cache.keys()) - all_session_ids:
        del session_cache[gone]

    write_state("done")


async def process_session(session):
    """Capture one session's content and schedule analysis if changed."""
    sid = session.session_id

    # Get screen content
    contents = await session.async_get_screen_contents()
    lines = []
    for i in range(contents.number_of_lines):
        line = contents.line(i)
        text = "".join(c.string for c in line.string_excluding_trailing_whitespace)
        lines.append(text)

    raw = "\n".join(lines[-80:])
    clean = strip_ansi(raw)
    h = content_hash(clean)

    # Get working directory
    path = str(Path.home())
    try:
        var = await session.async_get_variable("path")
        if var:
            path = var
    except Exception:
        pass

    project_info = await asyncio.get_event_loop().run_in_executor(
        None, get_project_info, path
    )
    project = project_info["name"]
    branch = project_info["branch"]
    title = session.name or project
    role_cfg = get_role_for_session(sid, project)

    # Check if content has changed
    cached = session_cache.get(sid)
    if cached and cached.content_hash == h:
        # No change — keep existing analysis, update role in case it changed
        cached.role = role_cfg.get("id", "builder") if isinstance(role_cfg, dict) and "id" in role_cfg else cached.role
        return

    # Content changed — create/update cache entry
    state = SessionState(
        session_id=sid,
        title=title,
        project=project,
        branch=branch,
        path=path,
        last_lines=clean,
        content_hash=h,
        role=_role_id_for(sid, project),
    )

    if not is_paused(clean):
        state.analysis = SessionAnalysis(action="continue")
        session_cache[sid] = state
        return

    # Paused — analyze
    state.analysis.ts_paused = state.analysis.ts_paused or now_iso()
    session_cache[sid] = state
    write_state("analyzing")

    analysis = await asyncio.get_event_loop().run_in_executor(
        None, analyze_with_claude, clean, project, branch, role_cfg
    )
    state.analysis.action      = analysis["action"]
    state.analysis.waiting_for = analysis["waiting_for"]
    state.analysis.options     = analysis["options"]
    state.analysis.recommended = analysis["recommended"]
    state.analysis.confidence  = analysis["confidence"]
    state.analysis.risk        = analysis["risk"]
    state.analysis.risk_reason = analysis["risk_reason"]
    state.analysis.decision    = analysis["decision"]

    # Auto-respond in semi/autopilot mode
    if mode in ("semi", "autopilot"):
        await maybe_auto_respond(session, state, role_cfg)


def _role_id_for(session_id: str, project: str) -> str:
    cfg = roles_config
    return (
        cfg.get("sessions", {}).get(session_id)
        or cfg.get("projects", {}).get(project)
        or cfg.get("global", "builder")
    )


def analyze_with_claude(content: str, project: str, branch: str, role: dict) -> dict:
    """Synchronous — runs in thread executor. Calls claude -p with Max sub flags."""
    role_prompt = role.get("prompt", "")
    threshold   = role.get("confidence_threshold", 0.80)

    base_prompt = PROMPT_PATH.read_text().strip() if PROMPT_PATH.exists() else DEFAULT_ANALYSIS_PROMPT
    analysis_prompt = (
        f"[ROLE: {role.get('name', 'Builder')}]\n{role_prompt}\n\n"
        + base_prompt
        .replace("{content}", content)
        .replace("{project}", project)
        .replace("{branch}", branch)
    )

    try:
        result = subprocess.run(
            [
                "claude", "-p", analysis_prompt,
                "--model",               "claude-haiku-4-5-20251001",
                "--effort",              "low",
                "--mcp-config",          '{"mcpServers":{}}',
                "--strict-mcp-config",
                "--disable-slash-commands",
                "--no-chrome",
                "--no-session-persistence",
            ],
            capture_output=True, text=True, timeout=ANALYSIS_TIMEOUT,
        )
        text = result.stdout.strip()
        match = re.search(r"\{[^{}]*\}", text, re.DOTALL)
        if match:
            parsed = json.loads(match.group())
            action = parsed.get("action", "pending")
            if action not in ("respond", "continue", "qa", "pending"):
                action = "pending"
            confidence = float(parsed.get("confidence", 0.0))
            # Risk check always wins
            risky, risk_reason = check_risk(content)
            if risky:
                action = "pending"
            # Confidence gate
            if action == "respond" and confidence < threshold:
                action = "pending"
            return {
                "action":      action,
                "waiting_for": str(parsed.get("summary", parsed.get("waiting_for", "")))[:100],
                "options":     [str(o) for o in parsed.get("options", ["y", "n"])][:8],
                "recommended": parsed.get("response") or parsed.get("recommended"),
                "confidence":  confidence,
                "risk":        risky or bool(parsed.get("risk", False)),
                "risk_reason": risk_reason or parsed.get("risk_reason"),
                "decision":    parsed.get("decision"),
            }
    except subprocess.TimeoutExpired:
        pass
    except Exception as e:
        print(f"[fleet] analysis error: {e}", flush=True)

    return {
        "action": "pending", "waiting_for": "Analysis failed",
        "options": ["y", "n"], "recommended": None,
        "confidence": 0.0, "risk": True,
        "risk_reason": "Analysis error", "decision": "Could not analyze",
    }


DEFAULT_ANALYSIS_PROMPT = """\
You are Fleet, supervising a Claude Code agent session. Keep it moving forward intelligently.

Terminal content (most recent last):
---
{content}
---
Project: {project} | Branch: {branch}

Choose ONE action:
"respond"  — agent is blocked waiting for input. Give the best response.
"continue" — agent is actively running. Leave it alone.
"qa"       — agent completed something meaningful. Flag for human review.
"pending"  — human judgment required: risky op, agent stuck, or not confident.

HARD RULES:
- Destructive ops (DROP TABLE, rm -rf, force push, reset --hard) → action=pending, risk=true
- Only use response options explicitly shown in the terminal
- If you can't identify valid options: action=pending

Return ONLY valid JSON, no markdown:
{"action":"respond","summary":"...","response":"y","options":["y","n"],"confidence":0.95,"risk":false,"risk_reason":null,"decision":null}
"""


async def maybe_auto_respond(session, state: SessionState, role: dict):
    """In semi/autopilot mode, auto-send if safe and confident."""
    a = state.analysis
    if a.action != "respond" or a.risk or state.blocked or not a.recommended:
        return
    if mode == "semi" and a.confidence < role.get("confidence_threshold", 0.80):
        return
    # Send response
    try:
        await session.async_send_text(a.recommended + "\n")
        a.auto_sent = a.recommended
        log_action(state.session_id, state.project, "respond", a.recommended, auto=True)
        print(f"[fleet] auto-sent '{a.recommended}' to {state.project}/{state.session_id[:8]}", flush=True)
    except Exception as e:
        print(f"[fleet] auto-respond failed: {e}", flush=True)


# ── Socket server ──────────────────────────────────────────────────────────────
async def run_socket_server(connection):
    """Unix domain socket server for Swift → daemon commands."""
    if SOCK_PATH.exists():
        SOCK_PATH.unlink()

    server = await asyncio.start_unix_server(
        lambda r, w: handle_socket_client(r, w, connection),
        path=str(SOCK_PATH),
    )
    print(f"[fleet] socket listening at {SOCK_PATH}", flush=True)
    async with server:
        await server.serve_forever()


async def handle_socket_client(reader, writer, connection):
    try:
        data = await reader.read(4096)
        cmd = json.loads(data.decode())
        result = await dispatch_command(cmd, connection)
        writer.write(json.dumps(result).encode())
        await writer.drain()
    except Exception as e:
        writer.write(json.dumps({"ok": False, "error": str(e)}).encode())
        await writer.drain()
    finally:
        writer.close()


async def dispatch_command(cmd: dict, connection) -> dict:
    global mode, roles_config
    action = cmd.get("cmd")

    if action == "approve":
        return await cmd_respond(cmd, connection, override_response=cmd.get("response"))

    elif action == "deny":
        return await cmd_respond(cmd, connection, override_response=cmd.get("response", "n"))

    elif action == "scan":
        asyncio.create_task(sweep(connection))
        return {"ok": True}

    elif action == "set_mode":
        mode = cmd.get("mode", "scout")
        write_state()
        return {"ok": True}

    elif action == "set_role":
        sid = cmd.get("session_id")
        proj = cmd.get("project")
        role = cmd.get("role", "builder")
        if sid:
            roles_config.setdefault("sessions", {})[sid] = role
        elif proj:
            roles_config.setdefault("projects", {})[proj] = role
        else:
            roles_config["global"] = role
        save_roles()
        # Update cached sessions
        for s in session_cache.values():
            s.role = _role_id_for(s.session_id, s.project)
        write_state()
        return {"ok": True}

    elif action == "status":
        return {"ok": True, "mode": mode, "sessions": len(session_cache)}

    return {"ok": False, "error": f"unknown command: {action}"}


async def cmd_respond(cmd: dict, connection, override_response: Optional[str] = None) -> dict:
    sid = cmd.get("session_id")
    state = session_cache.get(sid)
    if not state:
        return {"ok": False, "error": "session not found"}

    response = override_response or state.analysis.recommended or "y"
    app = await iterm2.async_get_app(connection)
    for window in app.windows:
        for tab in window.tabs:
            for session in tab.sessions:
                if session.session_id == sid:
                    await session.async_send_text(response + "\n")
                    state.analysis.auto_sent = response
                    state.analysis.action = "continue"
                    state.content_hash = ""  # Force re-analysis next sweep
                    log_action(sid, state.project, "respond", response, auto=False)
                    write_state()
                    return {"ok": True}

    return {"ok": False, "error": "iTerm2 session not found"}
```

```python
#!/usr/bin/env python3
"""
fleet.py — CLI control for Claude Fleet daemon.
All real work happens in fleet_daemon.py.
This is just a socket client + local utilities.
"""
from __future__ import annotations
import json
import os
import socket
import sys
from pathlib import Path

HOME = Path.home()
FLEET_DIR = HOME / ".claude-fleet"
SOCK_PATH = FLEET_DIR / "fleet.sock"
LOG_PATH = FLEET_DIR / "history.jsonl"


def send_command(cmd: dict) -> dict:
    """Send a command to the running daemon via Unix socket."""
    try:
        with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as s:
            s.connect(str(SOCK_PATH))
            s.sendall(json.dumps(cmd).encode())
            s.shutdown(socket.SHUT_WR)
            data = b""
            while chunk := s.recv(4096):
                data += chunk
            return json.loads(data.decode())
    except FileNotFoundError:
        return {"ok": False, "error": "Daemon not running. Start with: launchctl start com.wiking.claude-fleet"}
    except Exception as e:
        return {"ok": False, "error": str(e)}


def cmd_respond(args):
    sid = None
    text = None
    i = 0
    while i < len(args):
        if args[i] == "--session" and i + 1 < len(args):
            sid = args[i + 1]; i += 2
        elif args[i] == "--text" and i + 1 < len(args):
            text = args[i + 1]; i += 2
        else:
            i += 1
    if not sid or not text:
        print("Usage: fleet.py respond --session SESSION_ID --text RESPONSE")
        sys.exit(1)
    result = send_command({"cmd": "approve", "session_id": sid, "response": text})
    print(json.dumps(result, indent=2))


def cmd_status(_args):
    result = send_command({"cmd": "status"})
    print(json.dumps(result, indent=2))


def cmd_scan(args):
    mode = "scout"
    if "--mode" in args:
        idx = args.index("--mode")
        if idx + 1 < len(args):
            mode = args[idx + 1]
    result = send_command({"cmd": "scan"})
    print(json.dumps(result, indent=2))


def cmd_set_role(args):
    # fleet.py set-role --session ID --role builder
    # fleet.py set-role --project NAME --role cautious
    # fleet.py set-role --global builder
    cmd = {"cmd": "set_role"}
    i = 0
    while i < len(args):
        if args[i] == "--session" and i + 1 < len(args):
            cmd["session_id"] = args[i + 1]; i += 2
        elif args[i] == "--project" and i + 1 < len(args):
            cmd["project"] = args[i + 1]; i += 2
        elif args[i] == "--role" and i + 1 < len(args):
            cmd["role"] = args[i + 1]; i += 2
        elif args[i] == "--global" and i + 1 < len(args):
            cmd["role"] = args[i + 1]; i += 2
        else:
            i += 1
    result = send_command(cmd)
    print(json.dumps(result, indent=2))


def cmd_history(args):
    n = 20
    if "--last" in args:
        idx = args.index("--last")
        if idx + 1 < len(args):
            n = int(args[idx + 1])
    if not LOG_PATH.exists():
        print("No history yet.")
        return
    lines = LOG_PATH.read_text().strip().splitlines()
    for line in lines[-n:]:
        print(line)


def cmd_block(args):
    if not args:
        print("Usage: fleet.py block PROJECT_NAME")
        sys.exit(1)
    # Write to blocked.json directly (daemon reads on next sweep)
    blocked_path = FLEET_DIR / "blocked.json"
    try:
        blocked = json.loads(blocked_path.read_text()) if blocked_path.exists() else []
    except Exception:
        blocked = []
    if args[0] not in blocked:
        blocked.append(args[0])
    blocked_path.write_text(json.dumps(blocked, indent=2))
    print(f"Blocked: {args[0]}")


def cmd_unblock(args):
    if not args:
        print("Usage: fleet.py unblock PROJECT_NAME")
        sys.exit(1)
    blocked_path = FLEET_DIR / "blocked.json"
    try:
        blocked = json.loads(blocked_path.read_text()) if blocked_path.exists() else []
    except Exception:
        blocked = []
    blocked = [b for b in blocked if b != args[0]]
    blocked_path.write_text(json.dumps(blocked, indent=2))
    print(f"Unblocked: {args[0]}")


COMMANDS = {
    "respond":   cmd_respond,
    "status":    cmd_status,
    "scan":      cmd_scan,
    "set-role":  cmd_set_role,
    "history":   cmd_history,
    "block":     cmd_block,
    "unblock":   cmd_unblock,
}

if __name__ == "__main__":
    args = sys.argv[1:]
    if not args or args[0] not in COMMANDS:
        print("Usage: fleet.py <command> [args]")
        print("Commands:", ", ".join(COMMANDS))
        sys.exit(1)
    COMMANDS[args[0]](args[1:])
```

```swift
import Foundation

/// Sends commands to fleet_daemon.py over a Unix domain socket.
final class FleetSocket {
    static let shared = FleetSocket()
    private let socketPath = FileManager.default.homeDirectoryForCurrentUser
        .appendingPathComponent(".claude-fleet/fleet.sock").path

    func send(_ command: [String: Any], completion: ((Result<[String: Any], Error>) -> Void)? = nil) {
        DispatchQueue.global(qos: .userInitiated).async {
            do {
                let result = try self.sendSync(command)
                DispatchQueue.main.async { completion?(.success(result)) }
            } catch {
                DispatchQueue.main.async { completion?(.failure(error)) }
            }
        }
    }

    private func sendSync(_ command: [String: Any]) throws -> [String: Any] {
        let data = try JSONSerialization.data(withJSONObject: command)

        let fd = socket(AF_UNIX, SOCK_STREAM, 0)
        guard fd >= 0 else { throw FleetSocketError.connectionFailed }
        defer { close(fd) }

        var addr = sockaddr_un()
        addr.sun_family = sa_family_t(AF_UNIX)
        let pathBytes = socketPath.utf8CString
        withUnsafeMutableBytes(of: &addr.sun_path) { ptr in
            pathBytes.withUnsafeBytes { src in
                ptr.copyMemory(from: UnsafeRawBufferPointer(start: src.baseAddress, count: min(src.count, ptr.count)))
            }
        }

        let connectResult = withUnsafePointer(to: &addr) {
            $0.withMemoryRebound(to: sockaddr.self, capacity: 1) {
                Foundation.connect(fd, $0, socklen_t(MemoryLayout<sockaddr_un>.size))
            }
        }
        guard connectResult == 0 else { throw FleetSocketError.connectionFailed }

        _ = data.withUnsafeBytes { write(fd, $0.baseAddress, data.count) }
        shutdown(fd, SHUT_WR)

        var received = Data()
        var buf = [UInt8](repeating: 0, count: 4096)
        while true {
            let n = read(fd, &buf, buf.count)
            if n <= 0 { break }
            received.append(contentsOf: buf[..<n])
        }

        guard let response = try? JSONSerialization.jsonObject(with: received) as? [String: Any] else {
            throw FleetSocketError.badResponse
        }
        return response
    }

    // Convenience methods
    func approve(sessionId: String, response: String, completion: ((Bool) -> Void)? = nil) {
        send(["cmd": "approve", "session_id": sessionId, "response": response]) { result in
            completion?((try? result.get())?["ok"] as? Bool ?? false)
        }
    }

    func deny(sessionId: String, response: String, completion: ((Bool) -> Void)? = nil) {
        send(["cmd": "deny", "session_id": sessionId, "response": response]) { result in
            completion?((try? result.get())?["ok"] as? Bool ?? false)
        }
    }

    func scan() {
        send(["cmd": "scan"])
    }

    func setRole(sessionId: String? = nil, project: String? = nil, role: String) {
        var cmd: [String: Any] = ["cmd": "set_role", "role": role]
        if let sid = sessionId { cmd["session_id"] = sid }
        if let proj = project  { cmd["project"] = proj }
        send(cmd)
    }

    func setMode(_ mode: String) {
        send(["cmd": "set_mode", "mode": mode])
    }
}

enum FleetSocketError: LocalizedError {
    case connectionFailed
    case badResponse
    var errorDescription: String? {
        switch self {
        case .connectionFailed: return "Fleet daemon not running"
        case .badResponse: return "Invalid response from daemon"
        }
    }
}
```
